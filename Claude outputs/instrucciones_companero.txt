package com.example.proyectobd2.services;

import com.example.proyectobd2.dto.CreateColoniaRequest;
import com.example.proyectobd2.models.Colonia;
import com.example.proyectobd2.structures.LinkedList;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class ColoniaService {

    private final LinkedList<Colonia> colonias = new LinkedList<>();
    private Long siguienteId = 1L;

    public ColoniaService() {
        colonias.add(new Colonia(siguienteId++, "Centro", "Puebla", "72000",
                "Zona centro con red antigua", 19.0414, -98.2063, "activa"));
        colonias.add(new Colonia(siguienteId++, "La Paz", "Puebla", "72160",
                "Zona residencial con baja presion", 19.0560, -98.2290, "activa"));
        colonias.add(new Colonia(siguienteId++, "San Manuel", "Puebla", "72570",
                "Zona universitaria con cortes frecuentes", 19.0030, -98.1990, "activa"));
    }

    public List<Colonia> obtenerTodas() {
        return colonias.toList();
    }

    public Colonia obtenerPorId(Long id) {
        return colonias.find(c -> c.getId().equals(id));
    }

    public Colonia crear(CreateColoniaRequest request) {
        Colonia colonia = new Colonia(siguienteId++, request.nombreColonia(), request.municipio(),
                request.codigoPostal(), request.descripcion(), request.latitud(),
                request.longitud(), request.estado());
        colonias.add(colonia);
        return colonia;
    }
}

package com.example.proyectobd2.controllers;

import com.example.proyectobd2.dto.CreateColoniaRequest;
import com.example.proyectobd2.models.Colonia;
import com.example.proyectobd2.services.ColoniaService;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/colonias")
public class ColoniaController {

    private final ColoniaService coloniaService;

    public ColoniaController(ColoniaService coloniaService) {
        this.coloniaService = coloniaService;
    }

    @GetMapping
    public List<Colonia> obtenerTodas() {
        return coloniaService.obtenerTodas();
    }

    @GetMapping("/{id}")
    public ResponseEntity<Colonia> obtenerPorId(@PathVariable Long id) {
        Colonia colonia = coloniaService.obtenerPorId(id);
        if (colonia == null) return ResponseEntity.notFound().build();
        return ResponseEntity.ok(colonia);
    }

    @PostMapping
    public ResponseEntity<Colonia> crear(@RequestBody CreateColoniaRequest request) {
        return ResponseEntity.status(HttpStatus.CREATED).body(coloniaService.crear(request));
    }
}

 rama: feature/modulo-reportes

Uso real de la Queue: cada reporte nuevo entra a la COLA DE PENDIENTES.
El personal atiende los reportes en orden de llegada (FIFO) con
POST /api/reportes/atender, que saca el reporte mas antiguo de la cola
y le cambia el estado a "en atencion".

Endpoints extra:
  GET  /api/reportes/pendientes   -> lista la cola de pendientes
  POST /api/reportes/atender      -> atiende el siguiente (404 si no hay)

package com.example.proyectobd2.services;

import com.example.proyectobd2.dto.CreateReporteRequest;
import com.example.proyectobd2.models.Reporte;
import com.example.proyectobd2.structures.LinkedList;
import com.example.proyectobd2.structures.Queue;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.List;

@Service
public class ReporteService {

    private final LinkedList<Reporte> reportes = new LinkedList<>();
    private final Queue<Reporte> pendientes = new Queue<>();
    private Long siguienteId = 1L;

    public ReporteService() {
        registrar(new Reporte(siguienteId++, "Fuga en calle principal",
                "Sale agua de la banqueta desde hace dos dias",
                LocalDate.of(2026, 11, 10), LocalTime.of(9, 30), "Calle 5 de Mayo #120",
                "fuga1.jpg", "alta", "pendiente", 3L, 1L));
        registrar(new Reporte(siguienteId++, "Sin agua desde ayer",
                "No llega agua a la vivienda desde el lunes",
                LocalDate.of(2026, 11, 11), LocalTime.of(14, 15), "Av. Juarez #45",
                "sinagua.jpg", "media", "pendiente", 3L, 2L));
        registrar(new Reporte(siguienteId++, "Agua turbia",
                "El agua sale de color cafe",
                LocalDate.of(2026, 11, 12), LocalTime.of(8, 0), "Av. San Claudio #300",
                "turbia.jpg", "alta", "pendiente", 3L, 3L));
    }

    private void registrar(Reporte reporte) {
        reportes.add(reporte);
        pendientes.enqueue(reporte);
    }

    public List<Reporte> obtenerTodos() {
        return reportes.toList();
    }

    public Reporte obtenerPorId(Long id) {
        return reportes.find(r -> r.getId().equals(id));
    }

    public Reporte crear(CreateReporteRequest request) {
        Reporte reporte = new Reporte(siguienteId++, request.titulo(), request.descripcion(),
                LocalDate.now(), LocalTime.now().withNano(0), request.ubicacion(),
                request.fotografia(), request.nivelPrioridad(), "pendiente",
                request.usuarioId(), request.coloniaId());
        registrar(reporte);
        return reporte;
    }

    public List<Reporte> obtenerPendientes() {
        return pendientes.toList();
    }

    public Reporte atenderSiguiente() {
        Reporte reporte = pendientes.dequeue();
        if (reporte != null) {
            reporte.setEstado("en atencion");
        }
        return reporte;
    }
}

# AguaAlerta API

Sistema inteligente de monitoreo, deteccion de fugas y distribucion de
agua potable. Permite registrar colonias, depositos de agua y reportes
ciudadanos (fugas, falta de agua, contaminacion), y atender los reportes
en orden de llegada. API hecha con Spring Boot; en este parcial los datos
viven en memoria dentro de estructuras de datos propias.

## Integrantes
- Samuel Cuatzo Conde (samuuConde)
- NOMBRE DEL COMPANERO (Guaguen)

## Modulos y endpoints

| Modulo   | Metodo | Ruta                      | Descripcion                          |
|----------|--------|---------------------------|--------------------------------------|
| Health   | GET    | /api/health               | Responde OK                          |
| Health   | GET    | /api/health/db            | Prueba la conexion a PostgreSQL      |

| Reportes | POST   | /api/reportes/atender     | Atiende el reporte mas antiguo       |

## Estructuras de datos
- **LinkedList (propia, con nodos)** - paquete `structures`. Se usa en
  todos los services para guardar los registros en memoria.
- **Queue (propia, con nodos)** - paquete `structures`. Se usa en
  `ReporteService` como cola de reportes pendientes: se atienden en
  orden de llegada (FIFO).

Ambas tienen el metodo `toList()` para poder responder en JSON.

## Como correr el proyecto
1. Clonar el repositorio.
2. Copiar `.env.example` a `.env` y llenar las variables:
   DB_URL, DB_USERNAME, DB_PASSWORD
3. Correr:
   ./gradlew bootRun
   (en Windows: gradlew.bat bootRun)
4. La API queda en http://localhost:8080

## Postman
La coleccion esta en la carpeta `postman/` (formato Collection v2.1),
con la variable `baseUrl = http://localhost:8080`.
